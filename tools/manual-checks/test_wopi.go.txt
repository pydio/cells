package main

import (
    "encoding/json"
    "fmt"
    "net/http"
)

func main() {
    // Test WOPI discovery endpoint
    wopiURL := "http://localhost:8080/wopi"

    // Create a test request for WOPI discovery
    req, err := http.NewRequest("GET", wopiURL, nil)
    if err != nil {
        fmt.Println("Error creating request:", err)
        return
    }
    req.Header.Set("Accept", "application/json")

    client := &http.Client{Timeout: 10}
    resp, err := client.Do(req)
    if err != nil {
        fmt.Println("Error making request:", err)
        return
    }
    defer resp.Body.Close()

    if resp.StatusCode != http.StatusOK {
        fmt.Printf("Discovery request failed with status: %d\n", resp.StatusCode)
        return
    }

    // Parse discovery response
    var discovery map[string]interface{}
    if err := json.NewDecoder(resp.Body).Decode(&discovery); err != nil {
        fmt.Println("Error decoding discovery:", err)
        return
    }

    fmt.Println("WOPI Discovery successful!")
    if caps, ok := discovery["capabilities"]; ok {
        fmt.Printf("Capabilities: %v\n", caps)
    }
    if server, ok := discovery["server_url"]; ok {
        fmt.Printf("Server URL: %v\n", server)
    }
    if product, ok := discovery["product_name"]; ok {
        fmt.Printf("Product name: %v\n", product)
    }
}
